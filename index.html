Писать игры — просто

Никаких сложных настроек. Просто создай файл main.lua.

-- Пример: Создаем игрока через встроенный ECS
local tex = master.loadImage("player.png")
local batch = master.createSpriteBatch(tex)

local player = master.ecs.create({
    x = master.screen_w / 2, 
    y = master.screen_h / 2,
    
    update = function(self, dt)
        -- Логика движения здесь
    end,
    
    draw = function(self)
        batch:add(self.x, self.y, 64, 64, 1.0)
    end
})

function master.draw()
    master.clear(15, 10, 20)
    batch:clear()
    
    -- Движок сам обновит ECS и Сцены!
    
    batch:draw()
end
